<script is:inline>
  const firebaseConfig = {
    apiKey: "YOUR_API_KEY_HERE",
    authDomain: "YOUR_AUTH_DOMAIN_HERE",
    projectId: "YOUR_PROJECT_ID_HERE",
    storageBucket: "YOUR_STORAGE_BUCKET_HERE",
    messagingSenderId: "YOUR_MESSAGING_SENDER_ID_HERE",
    appId: "YOUR_APP_ID_HERE"
  };

  firebase.initializeApp(firebaseConfig);
  const db = firebase.firestore();

  function escapeHtml(str) {
    return String(str)
      .replaceAll('&', '&amp;')
      .replaceAll('<', '&lt;')
      .replaceAll('>', '&gt;')
      .replaceAll('"', '&quot;')
      .replaceAll("'", '&#39;');
  }

  function autoResize(textarea) {
    textarea.style.height = 'auto';
    textarea.style.height = textarea.scrollHeight + 'px';
  }

  function checkInput(textarea, btnId) {
    const btn = typeof btnId === 'string' ? document.getElementById(btnId) : btnId;
    btn.disabled = textarea.value.trim().length === 0;
  }

  function formatDate(date) {
    const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"];
    const d = date.getDate().toString().padStart(2, '0');
    const m = months[date.getMonth()];
    const y = date.getFullYear();
    const h = date.getHours().toString().padStart(2, '0');
    const min = date.getMinutes().toString().padStart(2, '0');
    return `${d} ${m} ${y} ${h}:${min}`;
  }

  async function submitPost() {
    const textarea = document.getElementById('post-content');
    const content = textarea.value.trim();
    if (!content) return;
    await db.collection('posts').add({ content, createdAt: new Date() });
    textarea.value = '';
    textarea.style.height = 'auto';
    document.getElementById('post-submit').disabled = true;
  }
  document.getElementById('post-submit').addEventListener('click', submitPost);

  async function submitComment(postId, textarea) {
    const content = textarea.value.trim();
    if (!content) return;
    await db.collection('posts').doc(postId).collection('comments').add({
      content: content,
      createdAt: new Date()
    });
    textarea.value = '';
    textarea.style.height = 'auto';
    textarea.nextElementSibling.disabled = true;
  }

  function renderPosts() {
    const postsDiv = document.getElementById('posts');
    db.collection('posts').orderBy('createdAt', 'asc').onSnapshot(snapshot => {
      snapshot.docChanges().forEach(change => {
        const doc = change.doc;
        if (change.type === 'added') {
          const post = doc.data();
          const postDiv = document.createElement('div');
          postDiv.className = 'post';
          postDiv.id = `post-${doc.id}`;
          const timeObj = post.createdAt ? post.createdAt.toDate() : new Date();
          
          postDiv.innerHTML = `
            <p style="margin:0; padding-right:25px; white-space: pre-wrap;">${escapeHtml(post.content)}</p>
            <small>${formatDate(timeObj)}</small>
            <div class="input-container" style="margin-top:10px;">
              <textarea class="reply-input" rows="1" placeholder="Reply..." 
                oninput="autoResize(this); checkInput(this, this.nextElementSibling)"></textarea>
              <button class="comment-submit-btn" disabled>Reply</button>
            </div>
            <div id="comments-${doc.id}"></div>
          `;

          const replyInput = postDiv.querySelector('.reply-input');
          const replyBtn = postDiv.querySelector('.comment-submit-btn');
          replyBtn.onclick = () => submitComment(doc.id, replyInput);

          postsDiv.prepend(postDiv);
          loadComments(doc.id);
        }
        if (change.type === 'removed') {
          document.getElementById(`post-${doc.id}`)?.remove();
        }
      });
    });
  }

  function loadComments(postId) {
    const container = document.getElementById(`comments-${postId}`);
    db.collection('posts').doc(postId).collection('comments').orderBy('createdAt', 'asc').onSnapshot(snap => {
      container.innerHTML = '';
      snap.forEach(doc => {
        const c = doc.data();
        const div = document.createElement('div');
        div.className = 'comment';
        div.id = `comment-${doc.id}`;
        const timeObj = c.createdAt ? c.createdAt.toDate() : new Date();
        div.innerHTML = `
          <p style="margin:0; padding-right:20px; white-space: pre-wrap;">${escapeHtml(c.content)}</p>
          <small>${formatDate(timeObj)}</small>
        `;
        container.appendChild(div);
      });
    });
  }

  renderPosts();
</script>
