                        .attr('dy', i === 0 ? initialDy : '1.1em') 
                        .text(line);
                });
            } else {
                text.attr('dy', '0.35em');
            }
        });

    // MODIFIED: Only show tooltip for nodes in the mainConcepts array
    node.on("mouseenter", function(event, d) {
        if (mainConcepts.includes(d.id)) {
            tooltipTitle.text(d.label.replace(/\n/g, ' '));
            tooltipDescription.text(d.description);
            tooltip.classed("visible", true);
        }
    })
    .on("mousemove", function(event) {
        if (d3.select(event.currentTarget).data()[0] && mainConcepts.includes(d3.select(event.currentTarget).data()[0].id)) {
            tooltip
                .style("left", (event.pageX + 15) + "px")
                .style("top", (event.pageY - 10) + "px");
        }
    })
    .on("mouseleave", function() {
        tooltip.classed("visible", false);
    });

    simulation.on("tick", () => {
        link
            .attr("x1", d => d.source.x)
            .attr("y1", d => d.source.y)
            .attr("x2", d => d.target.x)
            .attr("y2", d => d.target.y);

        node.attr("transform", d => `translate(${d.x},${d.y})`);
    });

    function dragstarted(event, d) {
        if (!event.active) simulation.alphaTarget(0.4).restart();
        if (d.id !== "personal-theory") { 
            d.fx = event.x; 
            d.fy = event.y; 
        }
    }
    function dragged(event, d) {
        if (d.id !== "personal-theory") {
            d.fx = event.x;
            d.fy = event.y;
        }
    }
    function dragended(event, d) {
        if (!event.active) simulation.alphaTarget(0); 
        if (d.id !== "personal-theory") { 
            d.fx = null;
            d.fy = null;
        }
    }

    window.addEventListener('resize', () => {
        const newWidth = container.clientWidth;
        svg.attr("width", newWidth);
        centerNode.fx = newWidth / 2;
        simulation.force("center", d3.forceCenter(newWidth / 2, height / 2));
        simulation.alpha(0.3).restart();
    });
</script>
