let currentAppIndex = 0;
let currentPageIndex = 0;

function renderContent(appIndex, pageIndex) {
const content = appContents[appIndex][pageIndex];
placeholderText.innerHTML = `<h2>${content.title}</h2>` + content.paragraphs.map(p => `<p>${p}</p>`).join('');
// Show/hide navigation buttons
prevPageBtn.style.display = pageIndex === 0 ? 'none' : 'inline-block';
nextPageBtn.style.display = pageIndex === appContents[appIndex].length - 1 ? 'none' : 'inline-block';
}

apps.forEach((app, index) => {
app.addEventListener('click', () => {
currentAppIndex = index;
currentPageIndex = 0;
ui.style.display = 'none';
placeholder.style.display = 'block';
renderContent(currentAppIndex, currentPageIndex);
});
});

closeButton.addEventListener('click', () => {
placeholder.style.display = 'none';
ui.style.display = 'block';
});

prevPageBtn.addEventListener('click', () => {
if (currentPageIndex > 0) {
currentPageIndex--;
renderContent(currentAppIndex, currentPageIndex);
}
});

nextPageBtn.addEventListener('click', () => {
if (currentPageIndex < appContents[currentAppIndex].length - 1) {
currentPageIndex++;
renderContent(currentAppIndex, currentPageIndex);
}
});

function updateTime() {
const now = new Date();
let hours = now.getHours();
const minutes = String(now.getMinutes()).padStart(2, '0');
const ampm = hours >= 12 ? 'PM' : 'AM';
hours = hours % 12 || 12;
document.getElementById('time').textContent = `${hours}:${minutes} ${ampm}`;
const days = ['Sun', 'Mon', 'Tue', 'Wed', 'Thu', 'Fri', 'Sat'];
const day = days[now.getDay()];
const date = `${day}. ${(now.getMonth() + 1).toString().padStart(2, '0')}/${now.getDate().toString().padStart(2, '0')}/${now.getFullYear()}`;
document.getElementById('date').textContent = date;
}
setInterval(updateTime, 1000);
updateTime();

function startPulse() {
powerButton.classList.add('pulse');
}
function stopPulse() {
powerButton.classList.remove('pulse');
}

powerButton.addEventListener('click', () => {
if (screenContent.classList.contains('off')) {
// Turning on
stopPulse();
powerButton.classList.remove('off');
logoContainer.style.display = 'block';
logoContainer.style.opacity = '1';
setTimeout(() => {
logoContainer.style.opacity = '0';
setTimeout(() => {
logoContainer.style.display = 'none';
screenContent.classList.remove('off');
screenContent.classList.add('on');
}, 1000);
}, 2000);
} else {
// Turning off
screenContent.classList.remove('on');
screenContent.classList.add('off');
powerButton.classList.add('off');
placeholder.style.display = 'none';
ui.style.display = 'block';
startPulse();
}
});

// Initialize pulse on page load if off
if (screenContent.classList.contains('off')) {
powerButton.classList.add('off');
startPulse();
} else {
stopPulse();
powerButton.classList.remove('off');
}
});
