## Thinking about context

The Hikvision Interactive Flat Panel is a concentrated assembly of many scarce minerals and metals. While the environmental price tag often feels weightless to global consumers, these elements are frequently tied to conflict-affected regions and hidden environmental costs.

![Hikvision in the wild](../../assets/images/blog/hikvisionclassroom.jpg)

<script is:inline>
  document.addEventListener('DOMContentLoaded', () => {
    const powerBtn = document.getElementById('power-button');
    const screen = document.getElementById('screen-content');
    const logo = document.getElementById('logo-container');
    const ui = document.getElementById('ui');
    const placeholder = document.getElementById('placeholder-screen');
    const appContentDiv = document.querySelector('.app-content');
    
    const appData = [
      { title: "Intro", body: "What's it cost to run a Hikvision Interactive Flat Panel in a Grade 2 Classroom? This journey explores the hidden environmental costs." },
      { title: "Assembly", body: "Manufacturing releases roughly 187,650 tons of CO2 annually. Metal smelting and plastics carry a heavy price." },
      { title: "Use", body: "Constant connectivity, updates, and teacher training add cognitive and energy costs to the classroom." },
      { title: "Summary", body: "No price tag accounts for the carbon and water embedded in this lifecycle. We must question the true value." }
    ];

    powerBtn.addEventListener('click', () => {
      if (screen.classList.contains('off')) {
        powerBtn.classList.remove('off');
        logo.style.display = 'block';
        setTimeout(() => logo.style.opacity = 1, 100);
        setTimeout(() => {
          logo.style.opacity = 0;
          screen.classList.replace('off', 'on');
        }, 2000);
      } else {
        screen.classList.replace('on', 'off');
        powerBtn.classList.add('off');
        placeholder.style.display = 'none';
      }
    });

    document.querySelectorAll('.app').forEach((app, i) => {
      app.addEventListener('click', () => {
        appContentDiv.innerHTML = `<h3>${appData[i].title}</h3><p>${appData[i].body}</p>`;
        placeholder.style.display = 'block';
      });
    });

    document.getElementById('close-button').addEventListener('click', () => {
      placeholder.style.display = 'none';
    });

    function updateTime() {
      const now = new Date();
      document.getElementById('time').textContent = now.toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'});
      document.getElementById('date').textContent = now.toLocaleDateString();
    }
    setInterval(updateTime, 1000);
    updateTime();
  });
</script>
