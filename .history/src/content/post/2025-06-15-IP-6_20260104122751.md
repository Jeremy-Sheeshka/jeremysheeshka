document.addEventListener('DOMContentLoaded', () => {
    const powerButton = document.getElementById('power-button');
    const screenContent = document.getElementById('screen-content');
    const timeDisplay = document.getElementById('time');
    const dateDisplay = document.getElementById('date');
    const apps = document.querySelectorAll('.app');
    const placeholderScreen = document.getElementById('placeholder-screen');
    const closeButton = document.getElementById('close-button');
    const placeholderText = document.querySelector('.placeholder-text.app-content');
    const prevPageBtn = document.getElementById('prev-page');
    const nextPageBtn = document.getElementById('next-page');

    let isOn = false;
    let currentAppIndex = 0;
    let currentPageIndex = 0;

    // --- DATA: FILL IN YOUR TEXT HERE FOR EACH APP ---
    const appContents = [
        // App 0: Intro
        [
            {
                title: "Intro",
                paragraphs: [
                    "This is the Intro section. (You can replace this text in the script tag below).",
                    "Lorem ipsum dolor sit amet, consectetur adipiscing elit. Nulla ac erat at dui placerat fermentum."
                ]
            }
        ],
        // App 1: Assembly
        [
            {
                title: "Assembly",
                paragraphs: [
                    "This is the Assembly section.",
                    "Details about materials, manufacturing, and labor costs go here."
                ]
            }
        ],
        // App 2: Use
        [
            {
                title: "Use",
                paragraphs: [
                    "This is the Use section.",
                    "Details about energy consumption and daily usage go here."
                ]
            }
        ],
        // App 3: Summary
        [
            {
                title: "Summary",
                paragraphs: [
                    "This is the Summary section.",
                    "Final thoughts and conclusions on the sustainability of the device."
                ]
            }
        ]
    ];

    function updateTime() {
        const now = new Date();
        timeDisplay.textContent = now.toLocaleTimeString('en-US', { hour: '2-digit', minute: '2-digit', hour12: false });
        dateDisplay.textContent = now.toLocaleDateString('en-US', { weekday: 'long', month: 'long', day: 'numeric' });
    }

    setInterval(updateTime, 1000);
    updateTime();

    // Toggle Power
    powerButton.addEventListener('click', () => {
        isOn = !isOn;
        if (isOn) {
            powerButton.classList.remove('off');
            screenContent.classList.add('on');
            screenContent.classList.remove('off');
        } else {
            powerButton.classList.add('off');
            screenContent.classList.remove('on');
            screenContent.classList.add('off');
            closeApp(); // Ensure app closes if power is cut
        }
    });

    // Render Content inside the App Screen
    function renderContent(appIndex, pageIndex) {
        if (!appContents[appIndex] || !appContents[appIndex][pageIndex]) return;
        
        const content = appContents[appIndex][pageIndex];
        
        // We use a custom div class "app-header-fake" instead of h2 to avoid the TOC # anchor generation
        let html = `<div class="app-header-fake">${content.title}</div>`;
        
        html += content.paragraphs.map(p => `<p>${p}</p>`).join('');
        
        placeholderText.innerHTML = html;

        // Handle Pagination Buttons
        const totalPages = appContents[appIndex].length;
        prevPageBtn.style.display = pageIndex === 0 ? 'none' : 'inline-block';
        nextPageBtn.style.display = pageIndex === totalPages - 1 ? 'none' : 'inline-block';
    }

    // Open App
    apps.forEach((app, index) => {
        app.addEventListener('click', () => {
            if (!isOn) return;
            currentAppIndex = index;
            currentPageIndex = 0;
            renderContent(currentAppIndex, currentPageIndex);
            placeholderScreen.style.display = 'flex';
        });
    });

    // Close App
    function closeApp() {
        placeholderScreen.style.display = 'none';
    }
    
    closeButton.addEventListener('click', closeApp);

    // Pagination Logic
    prevPageBtn.addEventListener('click', () => {
        if (currentPageIndex > 0) {
            currentPageIndex--;
            renderContent(currentAppIndex, currentPageIndex);
        }
    });

    nextPageBtn.addEventListener('click', () => {
        if (currentPageIndex < appContents[currentAppIndex].length - 1) {
            currentPageIndex++;
            renderContent(currentAppIndex, currentPageIndex);
        }
    });
});
</script>
